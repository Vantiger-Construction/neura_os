cmake_minimum_required(VERSION 3.14)
project(neura_offline_native_check LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The native chat owner is Windows-only")
endif()
if(NOT EXISTS "${FLUTTER_WRAPPER_INCLUDE}/flutter/method_channel.h")
  message(FATAL_ERROR "Provide the genuine Flutter Windows wrapper include directory")
endif()
add_library(neura_offline_owner_check OBJECT
  "../../windows/runner/neura_local_conversation.cpp")
target_include_directories(neura_offline_owner_check PRIVATE "${FLUTTER_WRAPPER_INCLUDE}")
target_compile_features(neura_offline_owner_check PRIVATE cxx_std_20)
target_compile_definitions(neura_offline_owner_check PRIVATE NOMINMAX)
if(MSVC)
  target_compile_options(neura_offline_owner_check PRIVATE /W4 /WX /EHsc)
endif()
